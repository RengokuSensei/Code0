    mesh.position.set(x, y, z);

  function radialTexture(stops, size = 256) {

    ctx.fillStyle = g;
    ctx.fillRect(0, 0, size, size);
    const tex = new THREE.CanvasTexture(c);
    tex.colorSpace = THREE.SRGBColorSpace;
    return tex;

    return new THREE.ShaderMaterial({

        uPower: { value: power },
        uStrength: { value: strength }
      },
      vertexShader: `
        varying vec3 vNormal;
        varying vec3 vViewDir;
        void main() {
          vNormal = normalize(normalMatrix * normal);
          vec4 mvPos = modelViewMatrix * vec4(position, 1.0);
          vViewDir = normalize(-mvPos.xyz);
          gl_Position = projectionMatrix * mvPos;

        uniform float uPower;
        uniform float uStrength;

    leftWrap.rotation.z = -0.62;
    body.add(leftWrap);
    armObjects.push({ wrap: leftWrap, restX: -1.32, restY: -0.72, restRz: -0.62, phase: 0 });

    const rightWrap = new THREE.Group();
    rightWrap.add(tplRight);
    rightWrap.position.set(1.32, -0.72, 0.15);
    rightWrap.rotation.z = 0.62;
    body.add(rightWrap);
    armObjects.push({ wrap: rightWrap, restX: 1.32, restY: -0.72, restRz: 0.62, phase: Math.PI * 0.85 });
  }

  // Procedural Fallback Builder
  function buildProceduralRobot() {
    const headSphereGeo = new THREE.SphereGeometry(1.0, 48, 48);
    const headMat = new THREE.MeshPhysicalMaterial({
      color: 0x282832,
      roughness: 0.26,
      clearcoat: 1.0,
      clearcoatRoughness: 0.15,
      metalness: 0.1,
      envMapIntensity: 0.95
    });
    const headMesh = new THREE.Mesh(headSphereGeo, headMat);
    const rimMesh = new THREE.Mesh(headSphereGeo, createRimMaterial(0x8a5cff, 4.5, 0.32));
    rimMesh.renderOrder = 2;
    headMesh.add(rimMesh);
    head.add(headMesh);

    // Ear Caps
    [-1.02, 1.02].forEach(ex => {
      const earGeo = new THREE.CylinderGeometry(0.24, 0.24, 0.14, 32);
      earGeo.rotateZ(Math.PI / 2);
      const earMat = new THREE.MeshPhysicalMaterial({ color: 0x1d1d23, roughness: 0.35, metalness: 0.3 });
      const ear = new THREE.Mesh(earGeo, earMat);
      ear.position.set(ex, 0, 0);
      head.add(ear);
    });

    addAntennaSpheres(head);
    mountEyesAndVisor(head, 1.0);

    // Procedural Arms
    const createArmGeom = () => {
      const g = new THREE.CapsuleGeometry(0.24, 0.45, 16, 24);
      const m = new THREE.MeshPhysicalMaterial({
        color: 0x1d1d23,
        roughness: 0.45,
        clearcoat: 0.3,
        clearcoatRoughness: 0.35
      });
      const arm = new THREE.Mesh(g, m);
      const armRim = new THREE.Mesh(g, createRimMaterial(0x8a5cff, 2.6, 1.05));
      armRim.renderOrder = 2;
      arm.add(armRim);
      return arm;
    };
    setupArms(createArmGeom(), createArmGeom());
  }

  // Load Remote GLBs with CORS
  const HEAD_GLB = "https://d8j0ntlcm91z4.cloudfront.net/user_38xzZboKViGWJOttwIXH07lWA1P/hf_20261018_172206_0181329a-86e2-41c9-8ab8-c496f29aaa53.glb";
  const ARM_GLB = "https://d8j0ntlcm91z4.cloudfront.net/user_38xzZboKViGWJOttwIXH07lWA1P/hf_20261018_172137_cd6086ec-07ef-44dc-bd15-95a450ddaa8c.glb";

  const loader = new GLTFLoader();

  Promise.allSettled([
    new Promise((resolve, reject) => loader.load(HEAD_GLB, resolve, undefined, reject)),
    new Promise((resolve, reject) => loader.load(ARM_GLB, resolve, undefined, reject))
  ]).then(results => {
    const headRes = results[0];
    const armRes = results[1];

    if (headRes.status === 'fulfilled' && armRes.status === 'fulfilled') {
      const headGltf = headRes.value;
      const armGltf = armRes.value;

      // Prepare Head
      const headModel = headGltf.scene;
      const headBox = new THREE.Box3().setFromObject(headModel);
      const headSize = headBox.getSize(new THREE.Vector3());
      const headCenter = headBox.getCenter(new THREE.Vector3());
      const headScale = 2.0 / (headSize.z || 2.0);
      headModel.scale.setScalar(headScale);
      headModel.position.set(-headCenter.x * headScale, -(headBox.min.y + 1.0) * headScale, -headCenter.z * headScale);

      prepareModel(headModel, { glossy: true, tint: 0x4a4a56, rim: { color: 0x8a5cff, power: 4.5, strength: 0.32 } });
      head.add(headModel);

      addAntennaSpheres(head);
      mountEyesAndVisor(head, 1.0);

      // Prepare Arms
      const armLeft = armGltf.scene;
      const armRight = armLeft.clone(true);
      [armLeft, armRight].forEach(a => {
        const aBox = new THREE.Box3().setFromObject(a);
        const aSize = aBox.getSize(new THREE.Vector3());
        const aCenter = aBox.getCenter(new THREE.Vector3());
        const aScale = 0.78 / (aSize.y || 1);
        a.scale.setScalar(aScale);
        a.position.set(-aCenter.x * aScale, -aCenter.y * aScale, -aCenter.z * aScale);
        prepareModel(a, { glossy: false, tint: 0x1d1d23, rim: { color: 0x8a5cff, power: 2.6, strength: 1.05 } });
      });

      setupArms(armLeft, armRight);
    } else {
      console.warn('[3D] GLB models unavailable or CORS error, activating procedural mascot fallback.');
      buildProceduralRobot();
    }
  }).catch(err => {
    console.warn('[3D] GLTF load error, activating procedural fallback:', err);
    buildProceduralRobot();
  });

  // Pointer Interaction
  const target = { x: 0, y: 0 };
  const look = { x: 0, y: 0 };
  let lastMoveTime = performance.now();

  function onPointerMove(e) {
    const rect = canvas.getBoundingClientRect();
    const clientX = e.clientX ?? (e.touches && e.touches[0] ? e.touches[0].clientX : rect.left + rect.width / 2);
    const clientY = e.clientY ?? (e.touches && e.touches[0] ? e.touches[0].clientY : rect.top + rect.height / 2);
    
    const rx = Math.max(W * 0.55, 300);
    const ry = Math.max(H * 0.55, 250);
    const cx = rect.left + rect.width * 0.5;
    const cy = rect.top + rect.height * 0.5;

    target.x = Math.max(-1.2, Math.min(1.2, (clientX - cx) / rx));
    target.y = Math.max(-1.2, Math.min(1.2, (clientY - cy) / ry));
    lastMoveTime = performance.now();
  }

  window.addEventListener('pointermove', onPointerMove, { passive: true });
  window.addEventListener('pointerdown', onPointerMove, { passive: true });
  window.addEventListener('touchmove', onPointerMove, { passive: true });

  // Blinking State
  let nextBlinkTime = performance.now() + 3000;
  let isBlinking = false;
  let blinkProgress = 0;

  // Animation Loop
  let lastTime = performance.now();
  let isVisible = true;

  // Sleep when scrolled out of view
  const io = new IntersectionObserver(([entry]) => {
    isVisible = entry.isIntersecting;
  }, { threshold: 0.1 });
  io.observe(canvas);

  function animate(now) {
    requestAnimationFrame(animate);
    if (!isVisible) return;

    const dt = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;
    const t = now * 0.001;

    // Idle look-around if pointer is still for > 3.5s
    if (now - lastMoveTime > 3500) {
      target.x = Math.sin(t * 0.45) * 0.45;
      target.y = Math.sin(t * 0.31 + 1.2) * 0.2;
    }

    // Smooth Look Decay
    const decay = 1.0 - Math.pow(0.0025, dt);
    look.x += (target.x - look.x) * decay;
    look.y += (target.y - look.y) * decay;

    // Harmonic Physics Bobbing
    const bob = Math.sin(t * 1.6) * 0.09;
    robot.position.y = bob;

    // Head & Body Rotations
    head.rotation.y = look.x * 0.52;
    head.rotation.x = look.y * 0.40;
    head.rotation.z = -look.x * 0.06 + Math.sin(t * 1.1) * 0.015;

    body.rotation.y = look.x * 0.25;
    body.rotation.x = look.y * 0.10;

    // Arms Floating Harmonic
    armObjects.forEach(arm => {
      const w = Math.sin(t * 2.0 + arm.phase);
      arm.wrap.position.y = arm.restY + w * 0.13 - bob * 0.35;
      arm.wrap.position.x = arm.restX + Math.sign(arm.restX) * look.x * -0.04;
      arm.wrap.rotation.z = arm.restRz + w * 0.12 * Math.sign(arm.restX);
      arm.wrap.rotation.x = Math.sin(t * 1.3 + arm.phase) * 0.08;
    });

    // Pupil Tracking
    const px = Math.max(-0.07, Math.min(0.07, look.x * 0.075));
    const py = Math.max(-0.06, Math.min(0.06, -look.y * 0.06));
    pupilWraps.forEach(pw => {
      pw.position.set(px, py, 0);
    });

    // Blinking Mechanics
    if (!isBlinking && now > nextBlinkTime) {
      isBlinking = true;
      blinkProgress = 0;
    }

    if (isBlinking) {
      blinkProgress += dt / 0.32;
      let scaleY = 1.0;
      if (blinkProgress < 0.5) {
        scaleY = 1.0 - (1.0 - 0.08) * (blinkProgress / 0.5);
      } else if (blinkProgress < 1.0) {
        scaleY = 0.08 + (1.0 - 0.08) * ((blinkProgress - 0.5) / 0.5);
      } else {
        isBlinking = false;
        scaleY = 1.0;
        nextBlinkTime = now + 2500 + Math.random() * 3500;
      }
      eyeObjects.forEach(eye => {
        eye.scale.y = scaleY;
      });
    }

    // Floor Glow follows Bobbing
    floorGlow.scale.set(3.4 - bob * 3.0, 0.42 - bob * 0.4, 1);
    floorMat.opacity = 0.75 + (0.09 - bob) * 1.5;

    // Render
    if (composer) {
      composer.render();
    } else {
      renderer.render(scene, camera);
    }
  }

  requestAnimationFrame(animate);

  // Resize Handling
  function onResize() {
    W = container.clientWidth || 400;
    H = container.clientHeight || 440;
    camera.aspect = W / H;
    camera.updateProjectionMatrix();
    renderer.setSize(W, H);
    if (composer) composer.setSize(W, H);
  }

  const ro = new ResizeObserver(onResize);
  ro.observe(container);
}
